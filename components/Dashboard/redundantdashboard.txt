// ─── Helpers ─────────────────────────────────────────────────────────────────

function fmt(n: number) {
  return `₱${n.toLocaleString()}`;
}

function pct(part: number, total: number) {
  if (!total) return "0.0";
  return ((part / total) * 100).toFixed(1);
}

const COIN_COLORS: Record<string, string> = {
  "₱1": "#64748b",
  "₱5": "#3b82f6",
  "₱10": "#10b981",
  "₱20": "#f59e0b",
};

const RANK_BADGE: Record<number, string> = {
  1: "bg-yellow-400 text-yellow-900",
  2: "bg-slate-300 text-slate-800",
  3: "bg-amber-600 text-amber-100",
};

// ─── Aggregate derived data (ALL summaries) ───────────────────────────────────

function useAggregate(summaries: DailySummaryWithId[]) {
  return useMemo(() => {
    if (!summaries.length) return null;

    const sorted = [...summaries].sort((a, b) => a.dateId.localeCompare(b.dateId));

    const totalRevenue = summaries.reduce((s, d) => s + d.totalRevenue, 0);
    const totalSales = summaries.reduce((s, d) => s + d.totalSales, 0);
    const totalDays = summaries.length;
    const avgDailyRevenue = Math.round(totalRevenue / totalDays);

    const allCoins = summaries.reduce(
      (acc, s) => ({
        coins_1: acc.coins_1 + s.coinDenomination.coins_1,
        coins_5: acc.coins_5 + s.coinDenomination.coins_5,
        coins_10: acc.coins_10 + s.coinDenomination.coins_10,
        coins_20: acc.coins_20 + s.coinDenomination.coins_20,
        totalValue: acc.totalValue + s.coinDenomination.totalValue,
      }),
      { coins_1: 0, coins_5: 0, coins_10: 0, coins_20: 0, totalValue: 0 }
    );

    const coinPieData = [
      { name: "₱1", value: allCoins.coins_1 * 1, count: allCoins.coins_1 },
      { name: "₱5", value: allCoins.coins_5 * 5, count: allCoins.coins_5 },
      { name: "₱10", value: allCoins.coins_10 * 10, count: allCoins.coins_10 },
      { name: "₱20", value: allCoins.coins_20 * 20, count: allCoins.coins_20 },
    ];

    const revenueTrend = sorted.map((s) => ({
      date: s.dateId.slice(5),
      fullDate: s.dateId,
      revenue: s.totalRevenue,
      sales: s.totalSales,
    }));

    const bestDay = [...summaries].sort((a, b) => b.totalRevenue - a.totalRevenue)[0];

    // Aggregate device performance across ALL days
    const deviceMap = new Map<
      string,
      {
        deviceId: string;
        branchId: string;
        totalRevenue: number;
        totalSales: number;
        days: number;
        coins_1: number;
        coins_5: number;
        coins_10: number;
        coins_20: number;
      }
    >();

    for (const s of summaries) {
      for (const d of s.rankedDevices) {
        const existing = deviceMap.get(d.deviceId);
        if (existing) {
          existing.totalRevenue += d.totalRevenue;
          existing.totalSales += d.salesCount;
          existing.days += 1;
          existing.coins_1 += d.coins_1;
          existing.coins_5 += d.coins_5;
          existing.coins_10 += d.coins_10;
          existing.coins_20 += d.coins_20;
        } else {
          deviceMap.set(d.deviceId, {
            deviceId: d.deviceId,
            branchId: d.branchId,
            totalRevenue: d.totalRevenue,
            totalSales: d.salesCount,
            days: 1,
            coins_1: d.coins_1,
            coins_5: d.coins_5,
            coins_10: d.coins_10,
            coins_20: d.coins_20,
          });
        }
      }
    }

    const allDevices = Array.from(deviceMap.values()).sort(
      (a, b) => b.totalRevenue - a.totalRevenue
    );

    // Aggregate branch performance across ALL days
    const branchMap = new Map<
      string,
      { branchId: string; totalRevenue: number; days: number; deviceCount: number }
    >();

    for (const s of summaries) {
      for (const b of s.branchSummary) {
        const existing = branchMap.get(b.branchId);
        if (existing) {
          existing.totalRevenue += b.totalRevenue;
          existing.days += 1;
        } else {
          branchMap.set(b.branchId, {
            branchId: b.branchId,
            totalRevenue: b.totalRevenue,
            days: 1,
            deviceCount: b.deviceCount,
          });
        }
      }
    }

    const branchBarData = Array.from(branchMap.values())
      .sort((a, b) => b.totalRevenue - a.totalRevenue)
      .map((b) => ({
        name: b.branchId.replace("branch_", "").slice(-6),
        fullId: b.branchId,
        revenue: b.totalRevenue,
        avgDaily: Math.round(b.totalRevenue / b.days),
      }));

    const bestBranch = branchBarData[0];

    return {
      totalRevenue,
      totalSales,
      totalDays,
      avgDailyRevenue,
      allCoins,
      coinPieData,
      revenueTrend,
      bestDay,
      allDevices,
      branchBarData,
      bestBranch,
    };
  }, [summaries]);
}

// ─── Per-day derived data ─────────────────────────────────────────────────────

function useDayDetail(summary: DailySummaryWithId | undefined) {
  return useMemo(() => {
    if (!summary) return null;

    const { coinDenomination, rankedDevices, branchSummary, unitBreakdown } = summary;

    const coinPieData = [
      { name: "₱1", value: coinDenomination.coins_1 * 1, count: coinDenomination.coins_1 },
      { name: "₱5", value: coinDenomination.coins_5 * 5, count: coinDenomination.coins_5 },
      { name: "₱10", value: coinDenomination.coins_10 * 10, count: coinDenomination.coins_10 },
      { name: "₱20", value: coinDenomination.coins_20 * 20, count: coinDenomination.coins_20 },
    ];

    const branchBarData = [...branchSummary]
      .sort((a, b) => b.totalRevenue - a.totalRevenue)
      .map((b) => ({
        name: b.branchId.replace("branch_", "").slice(-6),
        fullId: b.branchId,
        revenue: b.totalRevenue,
      }));

    const top5 = rankedDevices.slice(0, 5);

    const totalSalesCount = rankedDevices.reduce((s, d) => s + d.salesCount, 0);
    const avgSalesPerDevice =
      rankedDevices.length > 0
        ? (totalSalesCount / rankedDevices.length).toFixed(1)
        : "0";

    const bestBranch = [...branchSummary].sort(
      (a, b) => b.totalRevenue - a.totalRevenue
    )[0];

    const deviceEfficiency = unitBreakdown
      .filter((d) => d.salesCount > 0)
      .map((d) => ({ ...d, avgPerSale: Math.round(d.total / d.salesCount) }))
      .sort((a, b) => b.avgPerSale - a.avgPerSale);

    return { coinPieData, branchBarData, top5, avgSalesPerDevice, bestBranch, deviceEfficiency };
  }, [summary]);
}

function StatCard({
  title,
  value,
  sub,
  accent,
}: {
  title: string;
  value: string;
  sub?: string;
  accent?: string;
}) {
  return (
    <Card className="relative overflow-hidden">
      {accent && (
        <span className={`absolute left-0 top-0 h-full w-1 rounded-l-lg ${accent}`} />
      )}
      <CardHeader className="pb-1 pl-5">
        <CardDescription className="text-xs font-medium uppercase tracking-wider text-muted-foreground">
          {title}
        </CardDescription>
      </CardHeader>
      <CardContent className="pl-5">
        <p className="text-2xl font-bold">{value}</p>
        {sub && <p className="mt-0.5 text-xs text-muted-foreground">{sub}</p>}
      </CardContent>
    </Card>
  );
}

// ─── Main Dashboard ───────────────────────────────────────────────────────────

export default function BranchRanking() {
  const { summaries, loading, error } = useDailySummaries();
  const [selectedDate, setSelectedDate] = useState<string | null>(null);

  if (loading) return <LoadingSkeleton />;
  if (error)
    return (
      <div className="flex h-64 items-center justify-center text-destructive">
        Failed to load summaries: {error.message}
      </div>
    );
  if (!summaries.length)
    return (
      <div className="flex h-64 items-center justify-center text-muted-foreground">
        No summary data available.
      </div>
    );

  const latestDate = summaries[0]?.dateId;
  const selectedSummary = selectedDate
    ? summaries.find((s) => s.dateId === selectedDate)
    : null;

  return (
    <div className="min-h-screen bg-background p-6 text-foreground">
      {/* Header */}
      <div className="mb-6 flex items-start justify-between">
        <div>
          <h1 className="text-2xl font-bold tracking-tight">Coin Slot Dashboard</h1>
          <p className="text-sm text-muted-foreground">
            {summaries.length} days recorded · latest:{" "}
            <span className="font-medium text-foreground">{latestDate}</span>
          </p>
        </div>
        <div className="flex items-center gap-2">
          {selectedDate && (
            <Button variant="ghost" size="sm" onClick={() => setSelectedDate(null)} className="text-xs">
              ← All-Time Overview
            </Button>
          )}
          <Badge variant="outline" className="text-xs">
            {selectedDate ? `Viewing: ${selectedDate}` : "All-Time View"}
          </Badge>
        </div>
      </div>

      {/* Date selector pills */}
      <div className="mb-6 flex flex-wrap gap-2">
        <button
          onClick={() => setSelectedDate(null)}
          className={`rounded-full px-3 py-1 text-xs font-medium transition-colors ${
            !selectedDate
              ? "bg-primary text-primary-foreground"
              : "bg-muted text-muted-foreground hover:bg-muted/80"
          }`}
        >
          Overview
        </button>
        {summaries.map((s) => (
          <button
            key={s.dateId}
            onClick={() => setSelectedDate(s.dateId)}
            className={`rounded-full px-3 py-1 text-xs font-medium transition-colors ${
              selectedDate === s.dateId
                ? "bg-primary text-primary-foreground"
                : "bg-muted text-muted-foreground hover:bg-muted/80"
            }`}
          >
            {s.dateId}
          </button>
        ))}
      </div>

      <Separator className="mb-6" />

      {/* Content */}
      {!selectedDate ? (
        <OverviewTab summaries={summaries} />
      ) : selectedSummary ? (
        <DayDetailTab summary={selectedSummary} />
      ) : (
        <div className="flex h-32 items-center justify-center text-sm text-muted-foreground">
          No data for {selectedDate}
        </div>
      )}
    </div>
  );
}
